async function dpPollTask(taskId) {
    if (!taskId) return;
    
    // 先停止现有轮询
    dpStopPolling();
    
    isPolling = true;
    dpPollBadge.style.display = 'inline-block';
    dpGenerateBtn.disabled = true;
    pollAttempts = 0;
    dpStartProgress();
    dpShowGeneratingAnimation();
    
    // 存储当前任务ID，用于验证
    const currentTaskId = taskId;
    
    async function attemptPoll() {
        // 验证任务ID是否仍然有效
        if (!isPolling || currentTaskId !== taskId) {
            console.warn('轮询已被新任务取代');
            return;
        }
        
        pollAttempts++;
        if (pollAttempts > MAX_POLL_ATTEMPTS) {
            dpSetStatus('请求超时，请稍后查看', 'error');
            dpStopPolling();
            dpStopProgress();
            dpHideGeneratingAnimation();
            return;

            const { stop } = await dpHandleTaskResponse(data);
            
            if (stop) {
                dpStopPolling();
                return;
            }

                pollingTimer = setTimeout(attemptPoll, POLL_INTERVAL_MS);
            }
        } catch(error) {
            console.error('轮询错误:', error);
            dpSetStatus('查询出错，正在重试...', 'error');
            
            if (pollAttempts >= 8) {
                dpStopPolling();
                dpStopProgress();
                dpHideGeneratingAnimation();
                return;
            }
            
            if (isPolling && currentTaskId === taskId) {
                pollingTimer = setTimeout(attemptPoll, POLL_INTERVAL_MS);
            }
        }

    const increment = model.progressSpeed === 'slow' ? 0.3 : 1.5;
    let lastTime = 0;

        if (!lastTime) lastTime = now;
        const delta = now - lastTime;

            lastTime = now;

async function dpPollTask(taskId) {    
if (!taskId) return;        
// 先停止现有轮询

dpPollBadge.style.display = 'inline-block';    
dpGenerateBtn.disabled = true;

async function attemptPoll() {

console.warn('轮询已被新任务取代');            
return;        
}                
pollAttempts++;        
if (pollAttempts > MAX_POLL_ATTEMPTS) {            
dpSetStatus('请求超时，请稍后查看', 'error');            
dpStopPolling();            
dpStopProgress();            
dpHideGeneratingAnimation();            
return;

try {            
const data = await dpQueryTaskResult(currentTaskId);            
const { stop } = await dpHandleTaskResponse(data);                        
if (stop) {                
dpStopPolling();                
return;            
}                        
if (isPolling && currentTaskId === taskId) {                
pollingTimer = setTimeout(attemptPoll, POLL_INTERVAL_MS);            
}        
} catch(error) {            
console.error('轮询错误:', error);            
dpSetStatus('查询出错，正在重试...', 'error');                        
if (pollAttempts >= 8) {                
dpStopPolling();                
dpStopProgress();                
dpHideGeneratingAnimation();                
return;            
}                        
if (isPolling && currentTaskId === taskId) {                
pollingTimer = setTimeout(attemptPoll, POLL_INTERVAL_MS);            
}        
}    
}        
attemptPoll();
}
```

### 4. **进度条动画未正确清理**
**Bug位置：**
```javascript

**问题：** `requestAnimationFrame` 返回的ID可能被覆盖，导致无法取消。
**修复方法：**
```javascript
function dpStartProgress() {    
dpStopProgress();    
currentProgress = 0;    
dpProgressFill.style.width = '0%';    
dpProgressContainer.style.display = 'block';        
const model = MODELS[currentModelType];    
const increment = model.progressSpeed === 'slow' ? 0.3 : 1.5;    
let lastTime = 0;    
let animationId = null;        
function tick(now) {        
if (!lastTime) lastTime = now;        
const delta = now - lastTime;                
if (delta >= 40) {            
lastTime = now;            
if (currentProgress < PROGRESS_MAX) {                
currentProgress += Math.random() * increment + increment * 0.5;                
if (currentProgress > PROGRESS_MAX) currentProgress = PROGRESS_MAX;                
dpProgressFill.style.width = currentProgress + '%';            
}        
}                
if (currentProgress < PROGRESS_MAX) {            
animationId = requestAnimationFrame(tick);            
progressTimer = animationId; // 保持引用        
}    
}        
animationId = requestAnimationFrame(tick);    
progressTimer = animationId;
}
function dpStopProgress() {    
if (progressTimer) {        
cancelAnimationFrame(progressTimer);        
progressTimer = null;    
}    
dpProgressContainer.style.display = 'none';    
dpProgressFill.style.width = '0%';    
currentProgress = 0;
}
```
